# Examples for docs/config-reference.md, merged in by scripts/gen-config-reference.sh.
# Each `=== <directive>` starts a complete configuration; tests/integration/test_docs.py runs
# `bastion -t` on every one (paths below /srv/www, /etc/bastion and /var/... are mapped onto a
# scratch tree there). Keep them short and use the safe defaults: an example shows the directive,
# not a tuning recommendation. Run `./dev make gen-reference` after editing this file.
=== workers
workers auto;
server { listen 80; root /srv/www; }
=== user
user bastion;
server { listen 80; root /srv/www; }
=== pid
pid /run/bastion/bastion.pid;
server { listen 80; root /srv/www; }
=== error_log
error_log /var/log/bastion/error.log info;
server { listen 80; root /srv/www; }
=== include
include /etc/bastion/conf.d/*.conf;
=== access_log
access_log /var/log/bastion/access.log combined;
server {
    listen 80;
    root /srv/www;
    location /health { access_log off; return 200 "ok\n"; }
}
=== log_format
log_format short text "$remote_addr $request_method $request_uri $status";
access_log /var/log/bastion/access.log short;
server { listen 80; root /srv/www; }
=== worker_rlimit_nofile
worker_rlimit_nofile 65536;
server { listen 80; root /srv/www; }
=== shutdown_timeout
shutdown_timeout 10s;
server { listen 80; root /srv/www; }
=== open_file_cache
open_file_cache max=10000 ttl=60s;
server { listen 80; root /srv/www; }
=== server
server {
    listen 80;
    name example.com www.example.com;
    root /srv/www;
}
=== listen
server {
    listen 80;
    listen [::]:80;
    listen 443 tls http2;
    listen 443 http3;
    tls_cert /etc/bastion/tls/example.com.crt /etc/bastion/tls/example.com.key;
    root /srv/www;
}
=== name
server { listen 80; name example.com *.example.com; root /srv/www; }
server { listen 80 default_server; return 404; }
=== location
server {
    listen 80;
    root /srv/www;
    location /static/ { expires 7d; }
    location = /favicon.ico { access_log off; }
    location ~ \.php$ { return 404; }
}
=== root
server { listen 80; root /srv/www; }
=== alias
server {
    listen 80;
    root /srv/www;
    location /downloads/ { alias /srv/www/files/; }
}
=== index
server { listen 80; root /srv/www; index index.html index.htm; }
=== try_files
server {
    listen 80;
    root /srv/www;
    try_files $uri $uri/ /index.html;
}
=== autoindex
server {
    listen 80;
    root /srv/www;
    location /files/ { autoindex on; }

=== error_page
server {
    listen 80;
    root /srv/www;
    error_page 404 /404.html;
    error_page 500 502 503 504 /50x.html;
}
=== return

    listen 80;
    location /health { return 200 "ok\n"; }
    location /old { return 301 https://example.com/new; }
}
=== rewrite

    listen 80;
    root /srv/www;
    rewrite ^/blog/(\d+)$ /posts/$1 last;

server { listen 80; name example.com; redirect_https; }
=== internal

    root /srv/www;
    error_page 404 /404.html;

    email admin@example.com;
    ca letsencrypt;
    challenge http-01;

=== tls_profile

}
server {
    listen 443 tls http2;
    tls_cert /etc/bastion/tls/example.com.crt /etc/bastion/tls/example.com.key;
    tls_profile modern;
    root /srv/www;
}
=== security_headers
server {
    listen 443 tls http2;
    tls_cert /etc/bastion/tls/example.com.crt /etc/bastion/tls/example.com.key;
    security_headers basic;
    root /srv/www;
}
=== hsts
server {
    listen 443 tls http2;
    tls_cert /etc/bastion/tls/example.com.crt /etc/bastion/tls/example.com.key;
    security_headers basic;
    hsts "max-age=63072000; includeSubDomains";
    root /srv/www;
}
=== proxy
server {
    listen 80;
    location /api/ { proxy http://127.0.0.1:3000; }
    location / { root /srv/www; }
}
=== upstream
upstream app {
    server 127.0.0.1:3001;
    server 127.0.0.1:3002;
    server 127.0.0.1:3003 backup;
}
server { listen 80; proxy app; }
=== health_check
upstream app {
    server 127.0.0.1:3001;
    server 127.0.0.1:3002;
    health_check path=/healthz interval=5s fails=3 passes=2;
}
server { listen 80; proxy app; }
=== balance least_conn
upstream app {
    balance least_conn;
    server 127.0.0.1:3001;
    server 127.0.0.1:3002;
}
server { listen 80; proxy app; }

=== cache_store

=== cache
cache_store main { disk /var/cache/bastion size=1g; }

    listen 80;
    proxy http://127.0.0.1:3000;

}
=== cache_valid
cache_store main { disk /var/cache/bastion size=1g; }
server {
    listen 80;
    proxy http://127.0.0.1:3000;
    cache main;
    cache_valid 200 301 10m;
    cache_valid 404 1m;
}
=== cache_bypass
cache_store main { disk /var/cache/bastion size=1g; }

    listen 80;
    proxy http://127.0.0.1:3000;
    cache main;
    cache_bypass $cookie_session;
}
